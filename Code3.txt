from flask import Flask, jsonify
from datetime import datetime

app = Flask(__name__)

@app.route('/api/status', methods=['GET'])
def status():
    return jsonify({
        "service": "Flask API Demo",
        "status": "running",
        "time": datetime.now().isoformat()
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)
